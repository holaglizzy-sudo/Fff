local function HyphonBypass()
    local HandshakeFunctions = {}

    for _, Object in next, getgc(false) do
        if typeof(Object) == "function" and islclosure(Object) then
            local ok, DebugInfo = pcall(debug.info, Object, "s")
            if ok and DebugInfo then
                if DebugInfo:match("PlayerGui") and DebugInfo:match("%x+") then
                    local Constants = debug.getconstants(Object)
                    local UpValues  = debug.getupvalues(Object)
                    local uv = #UpValues
                    local co = #Constants

                    local isOriginal = (uv == 20 and co >= 140 and co <= 150)
                    local isNew      = (uv == 15 and co >= 143 and co <= 151)

                    if isOriginal or isNew then
                        table.insert(HandshakeFunctions, Object)
                    end
                end
            end
        end
    end

    warn("[Bypass] Closures encontradas: " .. #HandshakeFunctions)

    if #HandshakeFunctions == 0 then
        warn("Code 404 | No closures found | report in server")
        return
    end

    for Index, Function in next, HandshakeFunctions do
        if Index == 2 then
            hookfunction(Function, function(...)
                return table.unpack({})
            end)
        end
        if Index == 11 then
            hookfunction(Function, function(...)
                local Args    = {...}
                local NewArgs = table.clone(Args)
                NewArgs[2]    = false
                return table.unpack(NewArgs)
            end)
        end
    end

    Hyphon_Bypassed = true
end

    for part, props in pairs(NoclipParts) do

local Noclip = function()
    local Character = LocalPlayer.Character
    if not Character then return end
    local Root = Character:FindFirstChild("HumanoidRootPart")
    if not Root then return end
    local Pos = Root.Position
    local radius = 15
    local region = Region3.new(
        Pos - Vector3.new(radius, radius, radius),
        Pos + Vector3.new(radius, radius, radius)
    )
    local parts = Workspace:FindPartsInRegion3(region, nil, math.huge)
    for _, part in ipairs(parts) do
        if part:IsA("BasePart") and not ExclusionsSideWalk_Folder(part) and not Workspace:FindFirstChild("Do not touch") then
            if not NoclipParts[part] then
                NoclipParts[part] = { CanCollide = part.CanCollide }
                SetHiddenProperty(part, "CanCollide", false)
            end
        end
    end
end

local function Notify(title, message, duration)
    if Library and type(Library.Notification) == "function" then
        Library:Notification(title, message, duration)
        task.wait(duration)
        if Library.NotifHolder and Library.NotifHolder.Instance then
            for _, child in ipairs(Library.NotifHolder.Instance:GetChildren()) do
                if child:IsA("Frame") and child:FindFirstChild("Text") and child.Text and child.Text.Text == message then
                    child:Destroy()
                end
            end
        end
        return
    end

    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = title,
            Text = message,
            Duration = duration,
        })
    end)
end

local function FormatGoalCash(value)
    local formatted = tostring(math.floor(tonumber(value) or 0))
    local replacements
    repeat
        formatted, replacements = formatted:gsub("^(-?%d+)(%d%d%d)", "%1,%2")
    until replacements == 0
    return formatted
end

local function ReadCurrentCash()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    local main = playerGui and playerGui:FindFirstChild("Main")
    
    local hud = main and main:FindFirstChild("HUD")
    local background = hud and hud:FindFirstChild("Background")
    local moneyContainer = background and background:FindFirstChild("Money")
    
    local textObj = moneyContainer and moneyContainer:FindFirstChild("Amount")
    
    if not textObj and main then
        local money = main:FindFirstChild("Money")
        textObj = money and money:FindFirstChild("Amount")
    end
        
    if textObj and textObj.Text then
        local cleanedText = textObj.Text:gsub("[$,]", "")
        if cleanedText:find("%.") then
            cleanedText = cleanedText:split(".")[1]
        end
        return tonumber(cleanedText) or 0
    end
    return 0
end

local function GetLiveCharacter(timeout)
    local deadline = os.clock() + (timeout or 0)

    repeat
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Character and Character:FindFirstChild("HumanoidRootPart")

        if Character and Character.Parent and Humanoid and Humanoid.Health > 0 and RootPart and RootPart.Parent then
            return Character, Humanoid, RootPart
        end

        if not timeout or timeout <= 0 then
            break
        end

        task.wait(0.05)
    until os.clock() >= deadline

    return nil, nil, nil
end

local function moveToPosition(character, targetPosition, customStep, customWait)
    local _, _, humanoidRootPart = GetLiveCharacter(3)
    if not humanoidRootPart then return false end

    local moveStep = customStep or stepDistance
    local moveWait = customWait or waitTime
    local distance = (targetPosition - humanoidRootPart.Position).Magnitude

    while (Config.Enabled or CardConfig.Enabled) and distance > moveStep do
        _, _, humanoidRootPart = GetLiveCharacter(3)
        if not humanoidRootPart then return false end

        local currentPosition = humanoidRootPart.Position
        distance = (targetPosition - currentPosition).Magnitude
        if distance <= moveStep then
            break
        end

        local direction = (targetPosition - currentPosition).Unit
        humanoidRootPart.CFrame = CFrame.new(currentPosition + direction * moveStep)
        task.wait(moveWait)
    end

    _, _, humanoidRootPart = GetLiveCharacter(3)
    if humanoidRootPart then
        humanoidRootPart.CFrame = CFrame.new(targetPosition)
    else
        return false
    end
    return true
end

local function handleMovement(dirP)
    if RuntimeState.zzzMeth == "Bike" or RuntimeState.zzzMeth == "Bike TP" then
        local myBike = Workspace:FindFirstChild(LocalPlayer.Name .. "'s Car")
        if not myBike then
            local success = GetBike()
            if not success then return false end
            myBike = Workspace:FindFirstChild(LocalPlayer.Name .. "'s Car")
        end

        if myBike and myBike.PrimaryPart then
            myBike:SetPrimaryPartCFrame(CFrame.new(dirP.X, dirP.Y + 3, dirP.Z))
            task.wait(0.3)
            return true
        end
        return false
    else
        local Character, _, RootPart = GetLiveCharacter(3)
        if not Character or not RootPart then return false end

        local undergroundStart = Vector3.new(RootPart.Position.X, -2.4, RootPart.Position.Z)
        RootPart.CFrame = CFrame.new(undergroundStart)
        task.wait(0.5) 

        local undergroundTarget = Vector3.new(dirP.X, -2.4, dirP.Z)
        local success = moveToPosition(Character, undergroundTarget)

        if success then
            _, _, RootPart = GetLiveCharacter(3)
            if not RootPart then return false end
            RootPart.CFrame = CFrame.new(dirP)
            task.wait(0.5)
        end
        return success
    end
end

local function GetMarshmallowType(toolName)
    local lowerName = string.lower(toolName)
    local size = lowerName:match("^(small)") or lowerName:match("^(medium)") or lowerName:match("^(large)")
    if not size then return nil end
    if not (string.find(lowerName, "marshmallow", 1, true)
        or string.find(lowerName, "marhsmellow", 1, true)
        or string.find(lowerName, "marshmellow", 1, true)) then
        return nil
    end
    return string.upper(string.sub(size, 1, 1)) .. string.sub(size, 2) .. " Marshmallow Bag"
end

local function EquipTool(toolName)
    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local Tool = typeof(toolName) == "Instance" and toolName or GetTool(toolName)
    if not Humanoid or not Tool then return false end

    Humanoid:EquipTool(Tool)
    task.wait(0.2)
    return Tool.Parent == Character
end

local function SellAllMarshmallows()
    MarshmallowStep = "Selling marshmallows"
    handleMovement(Vector3.new(509, 4, 596))
    task.wait(0.5)

    local NPCs = Workspace:FindFirstChild("Folders") and Workspace.Folders:FindFirstChild("NPCs")
    local Lamont = NPCs and NPCs:FindFirstChild("Lamont Bell")
    local UpperTorso = Lamont and Lamont:FindFirstChild("UpperTorso")
    local SellPrompt = UpperTorso and UpperTorso:FindFirstChildWhichIsA("ProximityPrompt", true)
    
    if not UpperTorso or not SellPrompt then 
        Notify("marshmallow Alert", "Lamont Bell NPC or Sell Prompt missing....", 3)
        return 
    end

    local Character = LocalPlayer.Character
    local Backpack = LocalPlayer:FindFirstChild("Backpack")
    if not Character or not Backpack then return end

    local RootPart = GetRootPart(3)
    if RootPart then
        RootPart.CFrame = CFrame.lookAt(RootPart.Position, UpperTorso.Position)
    end

    while true do
        local itemsToSell = {}

        for _, tool in ipairs(Backpack:GetChildren()) do
            if tool:IsA("Tool") then
                local nameLower = string.lower(tool.Name)
                if string.find(nameLower, "marshmallow") or string.find(nameLower, "marhsmellow") or string.find(nameLower, "marshmellow") then
                    table.insert(itemsToSell, tool)
                end
            end
        end
        for _, tool in ipairs(Character:GetChildren()) do
            if tool:IsA("Tool") then
                local nameLower = string.lower(tool.Name)
                if string.find(nameLower, "marshmallow") or string.find(nameLower, "marhsmellow") or string.find(nameLower, "marshmellow") then
                    table.insert(itemsToSell, tool)
                end
            end
        end

        if #itemsToSell == 0 then break end

        for _, tool in ipairs(itemsToSell) do
            if EquipTool(tool) then
                FirePrompt(SellPrompt)
                local marshmallowType = GetMarshmallowType(tool.Name)
                if marshmallowType then
                    Config.Status.Earned = Config.Status.Earned + MarshmallowSellPrice[marshmallowType]
                    Config.Status.Total_Sold = Config.Status.Total_Sold + 1
                    Config.Status.Sold_Type[marshmallowType] = Config.Status.Sold_Type[marshmallowType] + 1
                end
                task.wait(0.3)
            else
                task.wait(0.1)
            end
        end
        task.wait(0.5)
    end
    task.wait(0.5)
end

local function RunMarshmallowFarm()
    local Map = Workspace:FindFirstChild("Map")
    local Apartments = Map and Map:FindFirstChild("APTS")
    if not Apartments then return end

    local OwnedApartment = GetPersonalApartment() or Get_House()

    if not OwnedApartment then
        local function Is_Vacant(Apartment)
            local Board = Apartment:FindFirstChild("Board")
            local Label = Board and Board:FindFirstChild("name") and Board.name:FindFirstChild("SurfaceGui") and Board.name.SurfaceGui:FindFirstChild("TextLabel")
            return Label and Label.Text == "VACANT"
        end

        local SelectedApartment = nil
        for _, ApartmentName in ipairs(Apartments_Lists) do
            local APT = Apartments:FindFirstChild(ApartmentName)
            if APT and APT:IsA("Model") and Is_Vacant(APT) then
                SelectedApartment = APT
                break
            end
        end

        if not SelectedApartment then
            Notify("marshmallow farm all apartments are owned pls rejoin a new server", 6)
            return
        end

        local Board = SelectedApartment:FindFirstChild("Board")
        if not Board then return end

        PreMaskTeleport(Board:GetPivot().Position, true)
        task.wait(2)

        local buyPrompt = Board:FindFirstChild("backboard") and Board.backboard:FindFirstChild("ProximityPrompt")
        if buyPrompt then

            end
        end

                CookAllAvailable(Interior, StovePosition)
            end
        end
        
        if Config.Enabled then
            SellAllMarshmallows()
        end

        task.wait(0.5)
    end
    MarshmallowStep = "Stopped"

local function cardToCFrame(value)

local function cardStepMove(character, targetPosition)
    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return false end

    local stepDistance = 1.5
    local waitTime = 0.08
    local currentPosition = root.Position
    local offset = targetPosition - currentPosition
    local distance = offset.Magnitude
    if distance <= 0.05 then
        root.CFrame = CFrame.new(targetPosition)
        return true
    end

    local direction = offset.Unit
    while distance > stepDistance and CardConfig.Enabled do
        currentPosition = currentPosition + direction * stepDistance
        root.CFrame = CFrame.new(currentPosition)
        task.wait(waitTime)
        offset = targetPosition - currentPosition
        distance = offset.Magnitude
    end

    root.CFrame = CFrame.new(targetPosition)
    return true
end

local function cardUndergroundTo(position)
    setupCardBaseplates()
    local character, root = getCardCharacter()
    if not character or not root then return false end
    local current = root.Position

    root.CFrame = CFrame.new(current.X, -2.4, current.Z)
    task.wait(1.8)

    if not cardStepMove(character, Vector3.new(position.X, -2.4, position.Z)) then return false end

    task.wait(1)
    character, root = getCardCharacter()
    if not character or not root then return false end
    root.CFrame = CFrame.new(position)
    return true
end

local function forceCardSpawn(destination)
    local selectedSpawnPoint = cardToCFrame(destination)
    local spawnActivated = true
    local characterAddedConnection

    local function applyCustomization(character)
        local bodyColors = LocalPlayer:FindFirstChild("BodyColors")
        if character:FindFirstChild("BodyColors") and bodyColors then
            character.BodyColors.HeadColor = bodyColors.HeadColor
            character.BodyColors.TorsoColor = bodyColors.TorsoColor
            character.BodyColors.LeftArmColor = bodyColors.LeftArmColor
            character.BodyColors.RightArmColor = bodyColors.RightArmColor
            character.BodyColors.LeftLegColor = bodyColors.LeftLegColor
            character.BodyColors.RightLegColor = bodyColors.RightLegColor
        end
    end

    local function teleportToSpawnPoint()
        local character = LocalPlayer.Character
        if selectedSpawnPoint and character and character:FindFirstChild("HumanoidRootPart") then
            if character.PrimaryPart then
                character:SetPrimaryPartCFrame(selectedSpawnPoint)
            else
                character.HumanoidRootPart.CFrame = selectedSpawnPoint
            end
        end
    end

    local function onCharacterAdded(character)
        character:WaitForChild("HumanoidRootPart")
        applyCustomization(character)
        if spawnActivated then
            wait(1.0)
            teleportToSpawnPoint()
        end
    end

    if characterAddedConnection then
        characterAddedConnection:Disconnect()
        characterAddedConnection = nil
    end

    characterAddedConnection = LocalPlayer.CharacterAdded:Connect(onCharacterAdded)
    task.delay(6, function()
        spawnActivated = false
        if characterAddedConnection then
            characterAddedConnection:Disconnect()
            characterAddedConnection = nil
        end
    end)

    return cardVPIDTeleport(selectedSpawnPoint)
end

    local targetPos = (typeof(target) == "Instance" and target:IsA("BasePart")) and target.Position or target
    Workspace.CurrentCamera.CFrame = CFrame.lookAt(root.Position + Vector3.new(0, 2.4, 0), targetPos)
end

local function findClosestActiveATM(character)
    local atmsFolder = Workspace:FindFirstChild("Map")
    atmsFolder = atmsFolder and atmsFolder:FindFirstChild("ATMS")
    if not atmsFolder then return end

    local root = character and character:FindFirstChild("HumanoidRootPart")
    if not root then return end

    local closestATM, shortestDist = nil, math.huge
    for _, atm in ipairs(atmsFolder:GetChildren()) do
        if atm:IsA("BasePart") and atm.Name:match("^ATM%d*$") then
            local prompt = atm:FindFirstChildWhichIsA("ProximityPrompt", true)
            if prompt and prompt.Enabled then
                local dist = (root.Position - atm.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    closestATM = atm
                end
            end
        end
    end

    return closestATM
end

local function waitForWalletIncrease(startCash, timeout)
    local highestCash = tonumber(startCash) or 0
    local started = tick()

    repeat
        task.wait(0.25)
        local currentCash = ReadCurrentCash()
        if currentCash > highestCash then
            highestCash = currentCash
        end
    until tick() - started >= (timeout or 5)

    return math.max(0, highestCash - (tonumber(startCash) or 0))
end

local function runCardFarmCycle()
    CardStep = "Buying Fake ID"
    local player = LocalPlayer
    local character, root, humanoid = getCardCharacter()
    if not character or not humanoid then return end

    pcall(function()
        game.GuiService.SelectedObject = nil
        if player.PlayerGui:FindFirstChild("ATM") then
            player.PlayerGui.ATM:Destroy()
        end
        humanoid:UnequipTools()
    end)
    clearOldCardFarmTools()
    task.wait(0.25)

    if not cardVPIDTeleport(CFrame.new(217, 4, -333)) then return end
    task.wait(0.5)

    local idSellerNPC = Workspace.Folders.NPCs:FindFirstChild("FakeIDSeller")
    if not idSellerNPC or not idSellerNPC:FindFirstChild("UpperTorso") then return end
    local idAttach = idSellerNPC.UpperTorso:FindFirstChild("Attachment")
    local idPrompt = idAttach and idAttach:FindFirstChildOfClass("ProximityPrompt")
    if not idPrompt then return end

    local idCounter = 0
    repeat
        if not CardConfig.Enabled then return end
        pointCamAt(idSellerNPC.UpperTorso)
        functionFirePrompt(idPrompt)
        task.wait(0.2)
        idCounter += 1
        if idCounter > 40 then return end
    until player.Backpack:FindFirstChild("Fake ID") or character:FindFirstChild("Fake ID")

    character, root, humanoid = getCardCharacter()
    if not character or not humanoid then return end
    local idTool = player.Backpack:FindFirstChild("Fake ID") or character:FindFirstChild("Fake ID")
    if idTool then
        local equipCounter = 0
        repeat
            pcall(function() humanoid:EquipTool(idTool) end)
            task.wait(0.15)
            character, root, humanoid = getCardCharacter()
            idTool = player.Backpack:FindFirstChild("Fake ID") or (character and character:FindFirstChild("Fake ID"))
            equipCounter += 1
        until not CardConfig.Enabled or (character and character:FindFirstChild("Fake ID")) or equipCounter >= 12
    end

    if not character or not character:FindFirstChild("Fake ID") then return end

    CardStep = "Applying at bank"
    if not cardUndergroundTo(Vector3.new(-51, 4, -315)) then return end
    task.wait(0.3)

    local applyCounter = 0
    local applicationConfirmed = false
    repeat
        if not CardConfig.Enabled then return end
        if bankApplicationSuccessful() then
            applicationConfirmed = true
            break
        end

        character, root, humanoid = getCardCharacter()
        if not character or not humanoid then return end

        local tellerNPC, tellerUpperTorso, tellerPrompt = getBankTellerPrompt()
        idTool = player.Backpack:FindFirstChild("Fake ID") or character:FindFirstChild("Fake ID")
        if idTool and idTool.Parent ~= character then
            pcall(function() humanoid:EquipTool(idTool) end)
            task.wait(0.1)
        end

        if tellerPrompt and tellerPrompt.Parent then
            pointCamAt(tellerUpperTorso)
            functionFirePrompt(tellerPrompt)
        elseif applyCounter % 5 == 0 then
            cardFastCFrame(CFrame.new(-70, 4, -318))
            task.wait(0.35)
            cardFastCFrame(CFrame.new(-51, 4, -315))
            task.wait(0.15)
            pressBankInteractKey()
        else
            pressBankInteractKey()
        end

        task.wait(1.0)
        applyCounter += 1
        if applyCounter > 18 then
            clearOldCardFarmTools()
            return
        end
    until bankApplicationSuccessful() or not (player.Backpack:FindFirstChild("Fake ID") or character:FindFirstChild("Fake ID"))

    if not applicationConfirmed and not bankApplicationSuccessful() then
        task.wait(11.5)
        if not bankApplicationSuccessful() then
            return
        end
    end

    CardStep = "Waiting for card"
    task.wait(35)
    cardFastCFrame(CFrame.new(-42, 4, -332))
    task.wait(0.05)
    if not forceCardSpawn(CFrame.new(-42, 4, -332)) then return end
    task.wait(0.15)

    local cardPickup = Workspace:FindFirstChild("CardPickup")

    local cardPickupAtt = cardPickup:FindFirstChild("Attachment")
    local cardPickupPrompt = cardPickupAtt and cardPickupAtt:FindFirstChildOfClass("ProximityPrompt")
    if not cardPickupPrompt then return end

    local pickupCounter = 0
    repeat
        if not CardConfig.Enabled then return end
        pointCamAt(cardPickup)
        functionFirePrompt(cardPickupPrompt)
        task.wait(0.04)
        pickupCounter += 1
        if pickupCounter > 180 then return end
    until player.Backpack:FindFirstChild("Card") or character:FindFirstChild("Card")

    character, root, humanoid = getCardCharacter()
    if not character or not humanoid then return end
    local cardTool = player.Backpack:FindFirstChild("Card") or character:FindFirstChild("Card")
    if cardTool then
        pcall(function() humanoid:EquipTool(cardTool) end)
        task.wait(0.03)
    end

    CardStep = "Swiping card"
    local closestATM = findClosestActiveATM(character)
    if not closestATM then return end

    local frontPos = closestATM.Position + (closestATM.CFrame.LookVector * -2.5)
    cardFastCFrame(CFrame.new(frontPos))
    task.wait(0.03)

    local atmPrompt = closestATM:FindFirstChildWhichIsA("ProximityPrompt", true)
    if not atmPrompt then return end

    local interactStart = tick()
    repeat
        if not CardConfig.Enabled then return end
        pointCamAt(closestATM)
        functionFirePrompt(atmPrompt)
        task.wait(0.08)
    until player.PlayerGui:FindFirstChild("ATM") or (tick() - interactStart > 8)

    if player.PlayerGui:FindFirstChild("ATM") then
        CardConfig.Status.Swipes += 1
        local cashBeforeSwipe = ReadCurrentCash()

        local swipeStart = tick()
        repeat
            if not CardConfig.Enabled then break end
            if not player.PlayerGui:FindFirstChild("ATM") then break end
            pointCamAt(closestATM)

            local atmFrame = player.PlayerGui.ATM:FindFirstChild("Frame")
            local swipeBtn = atmFrame and atmFrame:FindFirstChild("Swipe")
            if swipeBtn then game.GuiService.SelectedObject = swipeBtn end

            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
            task.wait(0.06)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
            task.wait(0.01)
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.BackSlash, false, game)
            task.wait(0.06)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.BackSlash, false, game)
            task.wait(0.1)
        until not player.PlayerGui:FindFirstChild("ATM") or not CardConfig.Enabled or (tick() - swipeStart > 12)

        local cardEarned = waitForWalletIncrease(cashBeforeSwipe, 5)
        CardConfig.Status.Earned += cardEarned
    end

    pcall(function()
        game.GuiService.SelectedObject = nil
    end)

    pcall(function()
        if player.PlayerGui:FindFirstChild("ATM") then
            VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.BackSlash, false, game)
            task.wait(0.05)
            VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.BackSlash, false, game)
        end
    end)

    character, root, humanoid = getCardCharacter()
    if humanoid then
        pcall(function() humanoid:UnequipTools() end)
    end
end

        local glowFrame = Instance.new("Frame")
        glowFrame.Size = UDim2.new(1.2, 0, 1.2, 0)
        glowFrame.Position = UDim2.new(-0.1, 0, -0.1, 0)
        glowFrame.BackgroundTransparency = 1
        glowFrame.BackgroundColor3 = Color3.fromRGB(0, 255, 255)
        glowFrame.BorderSizePixel = 0
        glowFrame.Parent = LogoImage.Instance
        
        local glowEffect = Instance.new("UIGradient")
        glowEffect.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 255, 255)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 200, 255)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 255, 255))
        })
        glowEffect.Parent = LogoImage.Instance
    end
    
    local mainFrame = Window.Instance
    if mainFrame then
        local glowBorder = Instance.new("UIStroke")
        glowBorder.Thickness = 3
        glowBorder.Color = Color3.fromRGB(0, 255, 255)
        glowBorder.Transparency = 0.3
        glowBorder.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        glowBorder.Parent = mainFrame
        
        task.spawn(function()
            while mainFrame and mainFrame.Parent do
                for i = 0.3, 0.8, 0.01 do
                    if not mainFrame.Parent then break end
                    glowBorder.Transparency = i
                    task.wait(0.02)
                end
                for i = 0.8, 0.3, -0.01 do
                    if not mainFrame.Parent then break end
                    glowBorder.Transparency = i
                    task.wait(0.02)
                end
            end
        end)
    end
    
    return Window
end

    while task.wait(0.3) do
        StatusLabel:SetText(Config.Enabled and "Status: Running" or "Status: Stopped")

        MediumLabel:SetText("Medium Sold: " .. Config.Status.Sold_Type["Medium Marshmallow Bag"])

        CardSwipesLabel:SetText("Cards Swiped: " .. CardConfig.Status.Swipes)
        
        local earned = Config.Status.Earned + CardConfig.Status.Earned
        EarnedLabel:SetText("Session Earned: " .. (earned < 0 and "-$" .. FormatGoalCash(math.abs(earned)) or "$" .. FormatGoalCash(earned)))
        DeathLabel:SetText("Deaths: " .. DiedCounter)

        local waterCount = CountTool("Water")
        local sugarCount = CountTool("Sugar Block Bag")
        local gelatinCount = CountTool("Gelatin")
        local totalIngredients = waterCount + sugarCount + gelatinCount
        
        local smallBags = CountTool("Small Marshmallow Bag")
        local mediumBags = CountTool("Medium Marshmallow Bag")
        local largeBags = CountTool("Large Marshmallow Bag")
        local totalBags = smallBags + mediumBags + largeBags
        
        local inventoryValue = (smallBags * 1470) + (mediumBags * 2840) + (largeBags * 4050)
        
        MarshStatusLabel:SetText("Status: " .. MarshmallowStep)
        MarshBagsLabel:SetText("Bags: " .. totalBags .. " | S" .. smallBags .. " / M" .. mediumBags .. " / L" .. largeBags)
        MarshIngredientsLabel:SetText("Ingredients: " .. totalIngredients .. " | Water " .. waterCount .. " / Sugar " .. sugarCount .. " / Gelatin " .. gelatinCount)
        MarshValueLabel:SetText("Inventory Value: $" .. FormatGoalCash(inventoryValue))

        local currentLiveWallet = ReadCurrentCash()
        GoalConfig.CurrentCash = currentLiveWallet

        if GoalConfig.SaveGoal and os.clock() - lastSave >= 6 then
            lastSave = os.clock()
            SaveGoalMemory()
        end

        if currentLiveWallet >= 1750000 then
            pcall(StopMarshmallowFarm)
            pcall(StopCardFarm)
            LocalPlayer:Kick("Goal System | Max cash limit reached.")
            return
        end

        if GoalConfig.Enabled and GoalConfig.CurrentCash > 0 and GoalConfig.CurrentCash >= GoalConfig.TargetCash then
            GoalConfig.Enabled = false
            pcall(StopMarshmallowFarm)
            pcall(StopCardFarm)
            task.wait(0.1)
            ClearGoalMemory()
            LocalPlayer:Kick("Goal System | Target reached! Balance: $" .. FormatGoalCash(GoalConfig.CurrentCash) .. " / Goal: $" .. FormatGoalCash(GoalConfig.TargetCash))
            return
        end
    end
end)
